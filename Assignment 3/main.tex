\section*{Question 1}
    \subsection*{Part 1}
    Let $C$ be a minimal cover for the knapsack set $K$. The corresponding valid inequality (cover cut) is:
    \[
        \sum_{i \in C} x_i \leq |C| - 1
    \]
    Let $C'$ be an extension of set $C$ by a single variable, defined as $C' = C \cup \{j\}$ for some $j \notin C$. 
    The cover cut for $C'$ is given by:
    \begin{align*}
        \sum_{i \in C'} x_i &= \sum_{i \in C} x_i + x_j \leq |C'| - 1 \\
        &= \sum_{i \in C} x_i + x_j \leq |C| \\
        \implies \quad \sum_{i \in C} x_i &\leq |C| - x_j
    \end{align*}
    Since $x_j \in \{0,1\}$, we observe that:
    \[
        \sum_{i \in C} x_i \leq |C|-1 \leq |C|-x_j
    \]
    Thus, the cover cut for $C$ dominates the cover cut for $C'$ because it imposes a more restrictive bound. 
    Consequently, the cover cut for $C'$ is weaker.
    
    \subsection*{Part 2}
    Lets lift a variable $x_k\notin C'$ into the cover cut for $C$. The lifted inequality is given by:
    \[
        \sum_{i \in C} x_i + \alpha_{k_1} x_k \leq |C| - 1
    \]
    Where the lifting coefficient $\alpha_{k_1}$ is defined as:
    \[
        \alpha_{k_1} = |C| - 1 - \max \left(\sum_{i \in C} x_i | x_k = 1, x \in \{0,1\}^n, \sum_{i \in N} a_i x_i \leq b, N=C\cup{k}\right)
    \]
    Similarly, we lift the same variable $x_k$ into the cover cut for $C'$. The lifted inequality can thus be expressed as:
    \[
        \sum_{i \in C'} x_i + \alpha_{k_2} x_k \leq |C'| - 1  \implies \sum_{i \in C} x_i + \alpha_{k_2} x_k \leq |C| - x_j
    \]
    Here the lifting coefficient $\alpha_{k_2}$ is defined as:
    \begin{align*}
        \alpha_{k_2} &= |C'| - 1 - \max \left(\sum_{i \in C'} x_i | x_k = 1, x \in \{0,1\}^n, \sum_{i \in N} a_i x_i \leq b, N=C'\cup{k}\right) \\
        &= |C| -  \max \left(\sum_{i \in C} x_i + x_j | x_k = 1, x \in \{0,1\}^n, \sum_{i \in N} a_i x_i \leq b, N=C\cup{j}\cup{k}\right)\\
        &\geq |C| - \max{x_j} - \max \left(\sum_{i \in C} x_i | x_k = 1, x \in \{0,1\}^n, \sum_{i \in N} a_i x_i \leq b, N=C\cup{k}\right) \\
        &\geq |C| - 1 - \max \left(\sum_{i \in C} x_i | x_k = 1, x \in \{0,1\}^n, \sum_{i \in N} a_i x_i \leq b, N=C\cup{k}\right) \\
        &\geq \alpha_{k_1}
    \end{align*}
    Thus we can see that:
    \begin{align*}
        \sum_{i \in C} x_i + \alpha_{k_2} x_k &\leq |C| - x_j \\
        \implies \sum_{i \in C} x_i + \alpha_{k_1} x_k \leq \sum_{i \in C} x_i + \alpha_{k_2} x_k &\leq |C| - x_j\\
        \implies \sum_{i \in C} x_i + \alpha_{k_1} x_k \leq \sum_{i \in C} x_i + \alpha_{k_2} x_k &\leq |C| - 1 \leq |C| - x_j
    \end{align*}
    
    Therefore, the lifted cover cut for $C$ dominates the lifted cover cut for $C'$, making the latter weaker.

\section*{Question 3}
We know that:
\[
    \sum_{i=1}^{n} a_i x_i + \sum_{g_i < 0} g_i y_i \leq \sum_{i=1}^{n} a_i x_i + \sum_{i=1}^{p} g_i y_i \leq b
\]

\begin{align*}
    \sum_{i=1}^{n} \left( \lfloor a_i \rfloor + f_i \right) x_i + \sum_{g_i < 0} g_i y_i  &\leq \lfloor b \rfloor + f_0 \\
    \implies \sum_{i=1}^{n} f_i x_i + \sum_{g_i < 0} g_i y_i - f_0  &\leq \lfloor b \rfloor - \sum_{i=1}^{n} \lfloor a_i \rfloor x_i
\end{align*}
Here we can clearly see that the RHS is an integer: 
\[
\color{red}{}\sum_{i=1}^{n} f_i x_i + \sum_{g_i < 0} g_i y_i - f_0  \leq K\\
\]

Thus let it be denoted as $K$ and lets see how it behaves for cases when $K \leq -1$ and $K \geq 0$.\\
Case $K\leq -1$:
\begin{align*}
    \sum_{i=1}^{n} \left(f_i - f_0\right)^+ x_i + \sum_{g_i < 0} g_i y_i \leq {\color{red}{}\sum_{i=1}^{n} f_i x_i + \sum_{g_i < 0} g_i y_i} &{\color{red}{}\leq K + f_0} \leq K\left(1-f_0\right)\\
    \implies \frac{1}{1-f_0} \left( \sum_{i=1}^{n}  \left(f_i - f_0\right)^+ x_i + \sum_{g_i < 0} g_i y_i \right) &\leq K\\
\end{align*}
This is because $ \left(f_i - f_0\right)^+ \leq f_i$, and also when $K \leq -1$ then $K + f_0 \leq K\left(1-f_0\right)$.\\
Case $K\geq 0$:
\begin{align*}
    \sum_{i=1}^{n} f_i x_i + \sum_{g_i < 0} g_i y_i - f_0  &\leq K \\
    \implies \sum_{i=1}^{n} \left(f_i-f_0\right)^+ x_i + \sum_{g_i < 0} g_i y_i &\leq K + f_0 \\
\end{align*}
Now lets split up the LHS based on the set $S = \{ i | f_i > f_0 \}$ and start with the subcase where $\sum_{i\in S} x_i \geq K+1$:
\begin{align*}
    \sum_{i=1}^{n} \left(f_i-f_0\right)^+ x_i + \sum_{g_i < 0} g_i y_i &= {\color{blue}{}\sum_{i \in S} \left(f_i-f_0\right) x_i + \sum_{g_i < 0} g_i y_i} \leq {\color{purple}{}\sum_{i \in S} f_i x_i + \sum_{g_i < 0} g_i y_i}\\
    \implies {\color{blue}{}\sum_{i \in S} f_i x_i + \sum_{g_i < 0} g_i y_i} &\leq K + f_0 {\color{blue}{}- f_0\left(\sum_{i \in S} x_i\right)} = K+f_0\left(1 - \sum_{i \in S} x_i\right)\\
    \implies {\color{purple}{}\sum_{i \in S} f_i x_i + \sum_{g_i < 0} g_i y_i} &\leq K+f_0\left(1 - \sum_{i \in S} x_i\right) \leq K + f_0\left(1 - (K+1)\right) = K(1-f_0)\\
    \implies \frac{1}{1-f_0} \left( \sum_{i=1}^{n} \left(f_i-f_0\right)^+ x_i + \sum_{g_i < 0} g_i y_i \right) &\leq K\\
\end{align*}
For the other subcase where $\sum_{i\in S} x_i \leq K$, we have:
\begin{align*}
    \sum_{i=1}^{n} \left(f_i-f_0\right)^+ x_i + \sum_{g_i < 0} g_i y_i &= \underbrace{{\color{blue}{}\sum_{i \in S} \left(f_i-f_0\right) x_i + \sum_{g_i < 0} g_i y_i} < \sum_{i \in S} \left(1-f_0\right) x_i + \sum_{g_i < 0} g_i y_i}_{f_i-f_0 < 1-f_0}\\
    \implies \frac{1}{1-f_0} \left( \sum_{i=1}^{n} \left(f_i-f_0\right)^+ x_i + \sum_{g_i < 0} g_i y_i \right) &< \sum_{i \in S} x_i + \underbrace{\frac{1}{1-f_0}\sum_{g_i < 0} g_i y_i}_{\text{Negative}} \leq \sum_{i \in S} x_i \leq K\\
    \implies \frac{1}{1-f_0} \left( \sum_{i=1}^{n} \left(f_i-f_0\right)^+ x_i + \sum_{g_i < 0} g_i y_i \right) &< K\\
\end{align*}
Thus we can conclude that the LHS:
\begin{align*}
    \frac{1}{1-f_0} \left( \sum_{i=1}^{n} \left(f_i-f_0\right)^+ x_i + \sum_{g_i < 0} g_i y_i \right) &\leq K &&\text{$\forall K\in \mathbb{Z}$}\\
\end{align*}
And thus substituting back for $K$ gives us the final valid inequality:
\begin{align*}
    \implies \frac{1}{1-f_0} \left( \sum_{i=1}^{n} \left(f_i-f_0\right)^+ x_i + \sum_{g_i < 0} g_i y_i \right) &\leq \left\lfloor b \right\rfloor - \sum_{i=1}^{n} \left\lfloor a_i \right\rfloor x_i\\
    \implies \sum_{i=1}^{n}\left( \lfloor a_i \rfloor  + \frac{\left(f_i-f_0\right)^+}{1-f_0} x_i \right) + \sum_{g_i < 0} \frac{g_i}{1-f_0} y_i &\leq \left\lfloor b \right\rfloor\\
\end{align*}
